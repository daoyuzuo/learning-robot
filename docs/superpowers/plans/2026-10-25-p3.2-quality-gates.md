```python
"""Quality report model and scoring helpers."""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Literal

Severity = Literal["error", "warning"]


@dataclass(frozen=True)
class QualityIssue:
    code: str
    message: str
    severity: Severity
    lesson_id: str | None = None
    concept_id: str | None = None
    resource_id: str | None = None

    def as_dict(self) -> dict:
        return {
            "code": self.code,
            "message": self.message,
            "severity": self.severity,
            "lesson_id": self.lesson_id,
            "concept_id": self.concept_id,
            "resource_id": self.resource_id,
        }


@dataclass(frozen=True)
class QualityMetric:
    key: str
    label: str
    earned: int
    possible: int
    detail: str

    def as_dict(self) -> dict:
        return {
            "key": self.key,
            "label": self.label,
            "earned": self.earned,
            "possible": self.possible,
            "detail": self.detail,
        }


def round_half_up(value: float) -> int:
    return int(value + 0.5)


def proportional(count: int, total: int, possible: int) -> int:
    if total <= 0 or possible <= 0:
        return 0
    return round_half_up(possible * (count / total))


def _issue_key(issue: QualityIssue) -> tuple:
    return (
        issue.code,
        issue.lesson_id or "",
        issue.concept_id or "",
        issue.resource_id or "",
        issue.message,
    )


@dataclass(frozen=True)
class QualityReport:
    score: int = 0
    metrics: list[QualityMetric] = field(default_factory=list)
    errors: list[QualityIssue] = field(default_factory=list)
    warnings: list[QualityIssue] = field(default_factory=list)
    provenance_evaluated: bool = False
    research_evaluated: bool = False
    earned: int = 0
    possible: int = 0

    @property
    def ok(self) -> bool:
        return not self.errors

    def as_dict(self) -> dict:
        return {
            "score": self.score,
            "earned": self.earned,
            "possible": self.possible,
            "ok": self.ok,
            "provenance_evaluated": self.provenance_evaluated,
            "research_evaluated": self.research_evaluated,
            "metrics": [metric.as_dict() for metric in self.metrics],
            "errors": [issue.as_dict() for issue in self.errors],
            "warnings": [issue.as_dict() for issue in self.warnings],
        }


def build_report(
    metrics: list[QualityMetric],
    errors: list[QualityIssue],
    warnings: list[QualityIssue],
    *,
    provenance_evaluated: bool,
    research_evaluated: bool,
) -> QualityReport:
    earned = sum(metric.earned for metric in metrics)
    possible = sum(metric.possible for metric in metrics)
    score = round_half_up(100 * earned / possible) if possible else 0
    return QualityReport(
        score=score,
        metrics=list(metrics),
        errors=sorted(errors, key=_issue_key),
        warnings=sorted(warnings, key=_issue_key),
        provenance_evaluated=provenance_evaluated,
        research_evaluated=research_evaluated,
        earned=earned,
        possible=possible,
    )
```

```python
"""Deterministic quality rules that only need pack content."""

from __future__ import annotations

from dataclasses import dataclass, field

from testbridge.content.repository import ContentPack
from testbridge.quality.report import QualityIssue, QualityMetric, proportional

LEVEL_ORDER = {"L1": 1, "L2": 2, "L3": 3, "L4": 4}


@dataclass
class StructuralResult:
    metrics: list[QualityMetric] = field(default_factory=list)
    errors: list[QualityIssue] = field(default_factory=list)
    warnings: list[QualityIssue] = field(default_factory=list)


def _error(code: str, message: str, **kwargs) -> QualityIssue:
    return QualityIssue(code=code, message=message, severity="error", **kwargs)


def _warning(code: str, message: str, **kwargs) -> QualityIssue:
    return QualityIssue(code=code, message=message, severity="warning", **kwargs)


def _level(value: str) -> int:
    return LEVEL_ORDER.get(value, 0)


def _module_concept_ids(pack: ContentPack) -> list[str]:
    seen: list[str] = []
    for module in sorted(pack.modules.values(), key=lambda item: (item.order, item.id)):
        for concept_id in module.concepts:
            if concept_id in pack.concepts and concept_id not in seen:
                seen.append(concept_id)
    return seen


def _lesson_exercises(pack: ContentPack, manifest) -> list:
    return [
        pack.exercises[exercise_id]
        for exercise_id in manifest.exercises
        if exercise_id in pack.exercises
    ]


def _max_level(exercises: list) -> int:
    return max((_level(item.level) for item in exercises), default=0)


def evaluate_structural(pack: ContentPack) -> StructuralResult:
    result = StructuralResult()
    lessons = pack.lessons_in_order()
    concept_ids = _module_concept_ids(pack)
    all_exercises = list(pack.exercises.values())

    covered = {concept_id for item in all_exercises for concept_id in item.concepts}
    missing = [concept_id for concept_id in concept_ids if concept_id not in covered]
    for concept_id in missing:
        result.errors.append(_error(
            "quality.concept_without_exercise",
            f"概念 {concept_id} 没有练习覆盖",
            concept_id=concept_id,
        ))

    for lesson in lessons:
        manifest = lesson.manifest
        exercises = _lesson_exercises(pack, manifest)
        if len(exercises) < 2:
            result.errors.append(_error(
                "quality.lesson_too_few_exercises",
                f"{manifest.id} 只有 {len(exercises)} 道练习，至少需要 2 道",
                lesson_id=manifest.id,
            ))
        lesson_concepts = set(manifest.concepts)
        for exercise in exercises:
            outside = [
                concept_id
                for concept_id in exercise.concepts
                if concept_id not in lesson_concepts
            ]
            if outside:
                result.errors.append(_error(
                    "quality.exercise_concept_outside_lesson",
                    f"{manifest.id} 的练习 {exercise.id} 引用了本课未声明的概念："
                    + "、".join(outside),
                    lesson_id=manifest.id,
                ))
        if exercises and _max_level(exercises) < _level(manifest.target_level):
            result.errors.append(_error(
                "quality.lesson_level_short",
                f"{manifest.id} 最高练习级别 L{_max_level(exercises)} "
                f"低于目标 {manifest.target_level}",
                lesson_id=manifest.id,
            ))
        if not manifest.resources:
            result.warnings.append(_warning(
                "quality.lesson_no_resource",
                f"{manifest.id} 没有资源",
                lesson_id=manifest.id,
            ))

    for concept_id in concept_ids:
        concept = pack.concepts[concept_id]
        levels = [
            _level(item.level)
            for item in all_exercises
            if concept_id in item.concepts
        ]
        if not levels:
            continue
        best = max(levels)
        if best < _level(concept.target_level):
            result.errors.append(_error(
                "quality.concept_level_short",
                f"概念 {concept_id} 最高练习级别 L{best} "
                f"低于目标 {concept.target_level}",
                concept_id=concept_id,
            ))

    covered_count = len(concept_ids) - len(missing)
    result.metrics.append(QualityMetric(
        "coverage.concepts", "概念有练习",
        proportional(covered_count, len(concept_ids), 15), 15,
        f"{covered_count}/{len(concept_ids)}",
    ))
    enough = sum(1 for lesson in lessons if len(lesson.manifest.exercises) >= 2)
    result.metrics.append(QualityMetric(
        "coverage.lesson_exercises", "每课至少 2 题",
        proportional(enough, len(lessons), 10), 10,
        f"{enough}/{len(lessons)}",
    ))
    types = len({item.type for item in all_exercises})
    result.metrics.append(QualityMetric(
        "coverage.type_variety", "题型多样",
        proportional(min(types, 3), 3, 5), 5,
        f"{types} 种",
    ))
    if types < 3:
        result.warnings.append(_warning(
            "quality.type_variety_low",
            f"全包只有 {types} 种题型，建议至少 3 种",
        ))

    lessons_level_ok = sum(
        1
        for lesson in lessons
        if _max_level(_lesson_exercises(pack, lesson.manifest))
        >= _level(lesson.manifest.target_level)
    )
    result.metrics.append(QualityMetric(
        "level.lessons", "课级达标",
        proportional(lessons_level_ok, len(lessons), 10), 10,
```

```python
"""Source provenance rules: URL traceability and research dossiers."""

from __future__ import annotations

import json
import math
from dataclasses import dataclass, field
from pathlib import Path
from urllib.parse import urlsplit

from testbridge.content.repository import ContentPack
from testbridge.quality.report import QualityIssue, QualityMetric, proportional

AB_TRUST = {"A", "B"}


@dataclass(frozen=True)
class Dossier:
    lesson_id: str
    queries: int
    hits: list[dict]


@dataclass
class ProvenanceResult:
    metrics: list[QualityMetric] = field(default_factory=list)
    errors: list[QualityIssue] = field(default_factory=list)
    warnings: list[QualityIssue] = field(default_factory=list)


def _error(code: str, message: str, **kwargs) -> QualityIssue:
    return QualityIssue(code=code, message=message, severity="error", **kwargs)


def _warning(code: str, message: str, **kwargs) -> QualityIssue:
    return QualityIssue(code=code, message=message, severity="warning", **kwargs)


def normalize_url(url: str) -> str:
    try:
        parts = urlsplit(url.strip())
        port = parts.port
    except ValueError:
        return ""
    scheme = parts.scheme.lower()
    if scheme not in {"http", "https"}:
        return ""
    host = (parts.hostname or "").lower()
    if not host:
        return ""
    if port and not (
        (scheme == "http" and port == 80) or (scheme == "https" and port == 443)
    ):
        host = f"{host}:{port}"
    path = parts.path.rstrip("/") or "/"
    return f"{scheme}://{host}{path}"


def _is_ab(hit: dict) -> bool:
    trust = hit.get("trust")
    if isinstance(trust, str):
        return trust in AB_TRUST
    try:
        return int(hit.get("authority")) >= 2
    except (TypeError, ValueError):
        return False


def load_dossiers(
    research_dir: Path,
) -> tuple[dict[str, Dossier], set[str], list[QualityIssue]]:
    dossiers: dict[str, Dossier] = {}
    malformed: set[str] = set()
    issues: list[QualityIssue] = []
    if not research_dir.is_dir():
        return dossiers, malformed, issues
    for path in sorted(research_dir.glob("*.json")):
        lesson_id = path.stem
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, UnicodeDecodeError, json.JSONDecodeError):
            data = None
        if not isinstance(data, dict) or not isinstance(data.get("hits"), list):
            malformed.add(lesson_id)
            issues.append(_error(
                "quality.research_malformed",
                f"{lesson_id} 调研档案无法解析",
                lesson_id=lesson_id,
            ))
            continue
        hits = [hit for hit in data["hits"] if isinstance(hit, dict)]
        queries = data.get("queries")
        dossiers[lesson_id] = Dossier(
            lesson_id=lesson_id,
            queries=len(queries) if isinstance(queries, list) else 0,
            hits=hits,
        )
    return dossiers, malformed, issues


def evaluate_provenance(
    pack: ContentPack,
    research_dir: Path | None = None,
    budget: int | None = None,
    *,
    include_metrics: bool = True,
) -> ProvenanceResult:
    result = ProvenanceResult()
    resources = list(pack.resources.values())

    trust_ok = sum(1 for resource in resources if resource.trust in AB_TRUST)
    metadata_ok = sum(
        1
        for resource in resources
        if resource.license and resource.last_verified and resource.why
    )
    for resource in resources:
        if resource.trust not in AB_TRUST:
            result.errors.append(_error(
                "quality.resource_trust",
                f"资源 {resource.id} 的 trust 是 {resource.trust}，只允许 A/B",
                resource_id=resource.id,
            ))
        if not (resource.license and resource.last_verified and resource.why):
            result.warnings.append(_warning(
                "quality.resource_metadata",
                f"资源 {resource.id} 缺少 license / last_verified / why",
                resource_id=resource.id,
            ))
    if include_metrics:
        result.metrics.append(QualityMetric(
            "provenance.trust", "资源可信",
            proportional(trust_ok, len(resources), 10), 10,
            f"{trust_ok}/{len(resources)}",
        ))
        result.metrics.append(QualityMetric(
            "provenance.metadata", "资源元数据完整",
            proportional(metadata_ok, len(resources), 5), 5,
            f"{metadata_ok}/{len(resources)}",
        ))

    if research_dir is None:
        return result

    dossiers, malformed, issues = load_dossiers(research_dir)
    result.errors.extend(issues)

    urls = {
        normalize_url(str(hit.get("url") or ""))
        for dossier in dossiers.values()
        for hit in dossier.hits
    }
    urls.discard("")
    traceable = 0
    for resource in resources:
        normalized = normalize_url(resource.url)
        if normalized and normalized in urls:
            traceable += 1
        else:
            result.errors.append(_error(
                "quality.resource_unsourced",
                f"资源 {resource.id} 的 URL 不在任何调研档案中",
                resource_id=resource.id,
            ))
    result.metrics.append(QualityMetric(
        "provenance.traceable", "资源可追溯",
        proportional(traceable, len(resources), 15), 15,
        f"{traceable}/{len(resources)}",
    ))

    lessons = pack.lessons_in_order()
    dossier_ok = 0
    query_ok = 0
    for lesson in lessons:
        lesson_id = lesson.manifest.id
        dossier = dossiers.get(lesson_id)
        if dossier is None:
            if lesson_id not in malformed:
                result.errors.append(_error(
                    "quality.research_missing",
                    f"{lesson_id} 没有调研档案",
                    lesson_id=lesson_id,
                ))
            continue
        if not dossier.hits:
            result.errors.append(_error(
                "quality.research_empty",
                f"{lesson_id} 的调研档案没有命中",
                lesson_id=lesson_id,
            ))
            continue
        dossier_ok += 1
        if not any(_is_ab(hit) for hit in dossier.hits):
            result.warnings.append(_warning(
                "quality.lesson_no_ab_source",
                f"{lesson_id} 没有 A/B 级来源",
                lesson_id=lesson_id,
            ))
        if budget is not None:
            threshold = max(1, math.ceil(budget / 2))
            if dossier.queries < threshold:
                result.warnings.append(_warning(
                    "quality.research_queries_low",
                    f"{lesson_id} 只查询了 {dossier.queries} 次，低于 {threshold}",
                    lesson_id=lesson_id,
                ))
            else:
                query_ok += 1

    result.metrics.append(QualityMetric(
        "research.dossiers", "调研可用",
        proportional(dossier_ok, len(lessons), 5), 5,
        f"{dossier_ok}/{len(lessons)}",
    ))
    if budget is not None:
        result.metrics.append(QualityMetric(
            "research.queries", "调研充分",
            proportional(query_ok, len(lessons), 5), 5,
            f"{query_ok}/{len(lessons)}",
        ))
    return result
```

```python
"""Combine structural and provenance checks into one quality report."""

from __future__ import annotations

from pathlib import Path

from testbridge.content.repository import ContentValidationError, load_pack
from testbridge.contract.validator import ValidationReport
from testbridge.quality.provenance import evaluate_provenance
from testbridge.quality.report import QualityReport, build_report
from testbridge.quality.structural import evaluate_structural


class QualityUnavailable(Exception):
    def __init__(self, report: ValidationReport) -> None:
        self.report = report
        super().__init__(f"content pack has {len(report.errors)} error(s)")


def evaluate_pack(
    pack_dir: Path,
    *,
    research_dir: Path | None = None,
    budget: int | None = None,
) -> QualityReport:
    try:
        pack = load_pack(pack_dir)
    except ContentValidationError as exc:
        raise QualityUnavailable(exc.report) from exc

    structural = evaluate_structural(pack)
    provenance = evaluate_provenance(
        pack,
        research_dir,
        budget,
        include_metrics=research_dir is not None,
    )
    return build_report(
        structural.metrics + provenance.metrics,
        structural.errors + provenance.errors,
        structural.warnings + provenance.warnings,
        provenance_evaluated=research_dir is not None,
        research_evaluated=research_dir is not None,
    )


def evaluate_draft(
    pack_dir: Path,
    *,
    research_dir: Path,
    budget: int,
) -> QualityReport:
    return evaluate_pack(pack_dir, research_dir=research_dir, budget=budget)
```

```python
def _run_quality(
    pack_dir: Path,
    research_dir: str | None,
    budget: int | None,
    as_json: bool,
) -> int:
    from testbridge.quality.evaluate import QualityUnavailable, evaluate_pack

    explicit_research = research_dir is not None
    research_path = Path(research_dir) if research_dir else None

    draft_path = pack_dir / "draft.json"
    if draft_path.is_file():
        try:
            draft = json.loads(draft_path.read_text(encoding="utf-8"))
        except (OSError, ValueError):
            draft = {}
        if research_path is None:
            research_path = pack_dir / "research"
        if budget is None and isinstance(draft.get("searches_per_lesson"), int):
            budget = draft["searches_per_lesson"]

    if explicit_research and (research_path is None or not research_path.is_dir()):
        print(f"ERROR research_dir: not found: {research_path}")
        return 2

    try:
        report = evaluate_pack(
            pack_dir, research_dir=research_path, budget=budget
        )
    except QualityUnavailable as exc:
        for issue in exc.report.errors:
            print(f"ERROR {issue.code}: {issue.message}")
        for issue in exc.report.warnings:
            print(f"WARN {issue.code}: {issue.message}")
        print("run tb validate first")
        return 2

    if as_json:
        print(json.dumps(report.as_dict(), ensure_ascii=False, indent=2))
    else:
        print(f"quality: {report.score}/100")
        for issue in report.errors:
            print(f"ERROR {issue.code}: {issue.message}")
        for issue in report.warnings:
            print(f"WARN {issue.code}: {issue.message}")
        print(
            "provenance: "
            + ("evaluated" if report.provenance_evaluated else "not evaluated")
        )
        print(
            "research: "
            + ("evaluated" if report.research_evaluated else "not evaluated")
        )
    return 0 if report.ok else 1
```
