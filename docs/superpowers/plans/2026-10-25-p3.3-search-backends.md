```python
"""OpenAlex works search adapter."""

from __future__ import annotations

import time
from typing import Any

import httpx

from testbridge.studio.search import SearchError, SearchHit

DEFAULT_BASE_URL = "https://api.openalex.org"
REVIEWED_TYPES = {"article", "review", "book-chapter", "proceedings-article"}
MAX_PER_PAGE = 50


def reconstruct_abstract(index: Any) -> str:
    if not isinstance(index, dict):
        return ""
    positions: dict[int, str] = {}
    for word, places in index.items():
        if not isinstance(places, list):
            continue
        for place in places:
            if isinstance(place, int) and place >= 0:
                positions[place] = str(word)
    if not positions:
        return ""
    return " ".join(positions[position] for position in sorted(positions))


def _doi_url(doi: Any) -> str:
    if not isinstance(doi, str) or not doi.strip():
        return ""
    value = doi.strip()
    if value.startswith(("http://", "https://")):
        return value
    if value.lower().startswith("doi:"):
        value = value[4:]
    return "https://doi.org/" + value


def _url_for(work: dict) -> str:
    doi_url = _doi_url(work.get("doi"))
    if doi_url:
        return doi_url
    primary = work.get("primary_location")
    if isinstance(primary, dict):
        landing = primary.get("landing_page_url")
        if isinstance(landing, str) and landing.startswith(("http://", "https://")):
            return landing
    open_access = work.get("open_access")
    if isinstance(open_access, dict):
        oa_url = open_access.get("oa_url")
        if isinstance(oa_url, str) and oa_url.startswith(("http://", "https://")):
            return oa_url
    work_id = work.get("id")
    if isinstance(work_id, str) and work_id.startswith(("http://", "https://")):
        return work_id
    return ""


def authority_for(work: dict) -> int:
    if work.get("is_retracted") is True:
        return -1
    work_type = str(work.get("type") or "").lower()
    if work_type in REVIEWED_TYPES:
        return 3
    if work_type == "preprint":
        return 2
    return 1


def _date_for(work: dict) -> str | None:
    date = work.get("publication_date")
    if isinstance(date, str) and date:
        return date
    year = work.get("publication_year")
    if isinstance(year, int) and not isinstance(year, bool):
        return str(year)
    if isinstance(year, str) and year.strip():
        return year.strip()
    return None


def _to_hit(work: dict) -> SearchHit | None:
    if not isinstance(work, dict):
        return None
    url = _url_for(work)
    if not url:
        return None
    abstract = reconstruct_abstract(work.get("abstract_inverted_index"))
    title = work.get("display_name") or work.get("title") or ""
    return SearchHit(
        title=str(title),
        url=url,
        snippet=abstract[:300],
        summary=abstract,
        content=abstract,
        authority=authority_for(work),
        date=_date_for(work),
    )


class OpenAlexSearch:
    def __init__(
        self,
        *,
        mailto: str | None = None,
        base_url: str = DEFAULT_BASE_URL,
        transport: httpx.BaseTransport | None = None,
        retry_delay: float = 1.0,
        timeout: float = 10.0,
    ) -> None:
        self.base_url = base_url.rstrip("/")
        self.mailto = mailto or None
        self.retry_delay = retry_delay
        self._client = httpx.Client(timeout=timeout, transport=transport)

    def _request(self, params: dict) -> dict:
        last_error = "openalex request failed"
        for attempt in range(2):
            try:
                response = self._client.get(self.base_url + "/works", params=params)
                if response.status_code < 400:
                    try:
                        data = response.json()
                    except ValueError:
                        raise SearchError("openalex returned non-JSON") from None
                    return data if isinstance(data, dict) else {}
                last_error = f"openalex status {response.status_code}"
                if response.status_code != 429 and response.status_code < 500:
                    break
            except (httpx.TimeoutException, httpx.HTTPError) as exc:
                last_error = type(exc).__name__
            if attempt == 0 and self.retry_delay:
                time.sleep(self.retry_delay)
        raise SearchError(last_error)

    def search(self, query: str, count: int = 10) -> list[SearchHit]:
        per_page = max(1, min(int(count), MAX_PER_PAGE))
        params: dict[str, Any] = {"search": query, "per-page": per_page}
        if self.mailto:
            params["mailto"] = self.mailto
        data = self._request(params)
        results = data.get("results")
        if not isinstance(results, list):
            return []
        hits = []
        for item in results:
            hit = _to_hit(item)
            if hit is not None:
                hits.append(hit)
        return hits[:per_page]

    def fetch(self, url: str) -> str:
        return ""
```

```python
"""Route search queries to a backend, with fallback only in auto mode."""

from __future__ import annotations

from dataclasses import dataclass

from testbridge.studio.search import SearchError, SearchHit, SearchProvider

ACADEMIC_KEYWORDS = (
    "论文", "综述", "研究", "数据集", "基准", "具身智能", "机器人学习",
    "强化学习", "模仿学习", "大模型", "基础模型", "扩散策略", "仿真到现实",
    "paper", "survey", "benchmark", "dataset", "state of the art", "sota",
    "vla", "vision-language-action", "embodied ai", "robot learning",
    "reinforcement learning", "imitation learning", "foundation model",
    "diffusion policy", "sim-to-real", "sim2real",
)
ACADEMIC_DOMAINS = frozenset({
    "ai", "ml", "research", "embodied-ai", "machine-learning",
    "computer-vision", "nlp",
})
MANUAL_BACKENDS = ("deepseek", "openalex")


@dataclass(frozen=True)
class SearchRoute:
    name: str
    provider: SearchProvider


def is_academic(
    topic: str,
    concept_titles: list[str],
    concept_domains: list[str],
) -> bool:
    text = " ".join([topic, *concept_titles]).lower()
    if any(keyword in text for keyword in ACADEMIC_KEYWORDS):
        return True
    return any(
        domain.strip().lower() in ACADEMIC_DOMAINS
        for domain in concept_domains
    )


class FallbackSearch:
    def __init__(self, primary: SearchProvider, secondary: SearchProvider) -> None:
        self.primary = primary
        self.secondary = secondary

    def search(self, query: str, count: int = 10) -> list[SearchHit]:
        try:
            hits = self.primary.search(query, count)
        except SearchError as primary_error:
            try:
                return self.secondary.search(query, count)
            except SearchError as secondary_error:
                raise SearchError(
                    f"primary failed: {primary_error}; "
                    f"secondary failed: {secondary_error}"
                ) from None
        if hits:
            return hits
        try:
            return self.secondary.search(query, count)
        except SearchError as secondary_error:
            raise SearchError(
                f"primary returned no hits; secondary failed: {secondary_error}"
            ) from None

    def fetch(self, url: str) -> str:
        try:
            content = self.primary.fetch(url)
        except SearchError:
            content = ""
        if content:
            return content
        try:
            return self.secondary.fetch(url)
        except SearchError:
            return ""


class SearchRouter:
    def __init__(self, providers: dict[str, SearchProvider]) -> None:
        self.providers = dict(providers)

    def route(
        self,
        backend: str,
        *,
        topic: str,
        concept_titles: list[str],
        concept_domains: list[str],
    ) -> SearchRoute:
        name = str(backend or "auto").strip().lower()
        if name in MANUAL_BACKENDS:
            provider = self.providers.get(name)
            if provider is None:
                raise SearchError(f"backend not configured: {name}")
            return SearchRoute(name, provider)
        return self._auto(topic, concept_titles, concept_domains)

    def _auto(
        self,
        topic: str,
        concept_titles: list[str],
        concept_domains: list[str],
    ) -> SearchRoute:
        primary_name = (
            "openalex" if is_academic(topic, concept_titles, concept_domains)
            else "deepseek"
        )
        secondary_name = (
            "deepseek" if primary_name == "openalex" else "openalex"
        )
        primary = self.providers.get(primary_name)
        secondary = self.providers.get(secondary_name)
        if primary is None and secondary is None:
            raise SearchError("no search backend configured")
        if primary is None:
            return SearchRoute(f"auto:{secondary_name}", secondary)
        if secondary is None:
            return SearchRoute(f"auto:{primary_name}", primary)
        return SearchRoute(
            f"auto:{primary_name}->{secondary_name}",
            FallbackSearch(primary, secondary),
        )
```
