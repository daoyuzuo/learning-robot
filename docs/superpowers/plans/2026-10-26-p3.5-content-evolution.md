```python
"""Snapshots and comparisons for content evolution."""

from __future__ import annotations

import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

import yaml

from testbridge.contract.loader import load_lesson_file
from testbridge.quality.evaluate import QualityUnavailable, evaluate_draft
from testbridge.studio.outline import LessonPlan
from testbridge.studio.store import DraftStore

LEVEL_ORDER = {"L1": 1, "L2": 2, "L3": 3, "L4": 4}
EMPTY_METRICS = {
    "exercise_count": 0, "types": [], "levels": [], "max_level": "",
    "resource_count": 0, "trust": {}, "html_hash": "",
}


def _yaml_list(store: DraftStore, draft_id: str, relative: str, key: str) -> list:
    try:
        data = yaml.safe_load(store.read(draft_id, relative))
    except FileNotFoundError:
        return []
    if not isinstance(data, dict):
        return []
    return list(data.get(key) or [])


def _read_lesson(store, draft_id, plan):
    path = (
        store.path(draft_id) / "modules" / plan.module_id
        / "lessons" / f"{plan.lesson_id}.html"
    )
    if not path.is_file():
        return "", [], [], EMPTY_METRICS
    html = path.read_text(encoding="utf-8")
    manifest, _ = load_lesson_file(path)
    exercises = [
        item for item in _yaml_list(store, draft_id, "exercises.yaml", "exercises")
        if item.get("id") in manifest.exercises
    ]
    resources = [
        item for item in _yaml_list(store, draft_id, "resources.yaml", "resources")
        if item.get("id") in manifest.resources
    ]
    levels = [str(item.get("level") or "") for item in exercises]
    types = sorted({str(item.get("type") or "") for item in exercises if item.get("type")})
    trust: dict[str, int] = {}
    for item in resources:
        name = str(item.get("trust") or "?")
        trust[name] = trust.get(name, 0) + 1
    metrics = {
        "exercise_count": len(exercises),
        "types": types,
        "levels": levels,
        "max_level": max(levels, key=lambda value: LEVEL_ORDER.get(value, 0), default=""),
        "resource_count": len(resources),
        "trust": trust,
        "html_hash": hashlib.sha256(html.encode("utf-8")).hexdigest()[:16],
    }
    return html, exercises, resources, metrics


def lesson_metrics(store, draft_id, plan) -> dict:
    return _read_lesson(store, draft_id, plan)[3]


def snapshot_lesson(store, draft_id, plan) -> dict:
    html, exercises, resources, metrics = _read_lesson(store, draft_id, plan)
    draft = store.load(draft_id)
    entry = draft.get("steps", {}).get("lessons", {}).get(plan.lesson_id, {})
    version = entry.get("prompt_version") or draft.get("prompt_version") or "studio/1"
    return {
        "schema": "studio-snapshot/1",
        "draft_id": draft_id,
        "lesson_id": plan.lesson_id,
        "module_id": plan.module_id,
        "lesson_title": plan.title,
        "created_at": datetime.now(timezone.utc).isoformat(),
        "prompt_version": version,
        "html": html,
        "exercises": exercises,
        "resources": resources,
        "metrics": metrics,
    }


def _stamp(created_at: str) -> str:
    try:
        value = datetime.fromisoformat(created_at)
        if value.tzinfo is None:
            value = value.replace(tzinfo=timezone.utc)
        return value.astimezone(timezone.utc).strftime("%Y%m%dT%H%M%S%fZ")
    except ValueError:
        return datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S%fZ")


def save_snapshot(store, snapshot: dict) -> str:
    relative = (
        f"history/{snapshot['lesson_id']}/{_stamp(snapshot['created_at'])}.json"
    )
    store.write(
        snapshot["draft_id"], relative,
        json.dumps(snapshot, ensure_ascii=False, indent=2),
    )
    return relative


def latest_snapshot(store, draft_id, lesson_id) -> dict | None:
    root = store.path(draft_id) / "history" / lesson_id
    if not root.is_dir():
        return None
    files = sorted(root.glob("*.json"))
    if not files:
        return None
    return json.loads(files[-1].read_text(encoding="utf-8"))


def compare_snapshots(before: dict, after: dict) -> dict:
    old = before.get("metrics") or EMPTY_METRICS
    new = after.get("metrics") or EMPTY_METRICS
    old_types = set(old.get("types") or [])
    new_types = set(new.get("types") or [])
    return {
        "html": old.get("html_hash") != new.get("html_hash"),
        "exercise_count": new.get("exercise_count", 0) - old.get("exercise_count", 0),
        "types_added": sorted(new_types - old_types),
        "types_removed": sorted(old_types - new_types),
        "max_level_before": old.get("max_level", ""),
        "max_level_after": new.get("max_level", ""),
        "resource_count": new.get("resource_count", 0) - old.get("resource_count", 0),
        "prompt_version": before.get("prompt_version") != after.get("prompt_version"),
    }


def draft_quality(store, draft_id) -> dict | None:
    draft = store.load(draft_id)
    try:
        report = evaluate_draft(
            store.path(draft_id),
            research_dir=store.path(draft_id) / "research",
            budget=int(draft.get("searches_per_lesson") or 1),
        )
    except QualityUnavailable:
        return None
    return {"score": report.score, "possible": report.possible, "ok": report.ok}
```

In `evolution.py`:

```python
import copy
import threading

class EvolutionBusy(Exception):
    """An evolution job is already running."""


class EvolutionJob:
    def __init__(self, store: DraftStore, runner) -> None:
        self.store = store
        self.runner = runner
        self._lock = threading.Lock()
        self._state = {
            "status": "idle", "total": 0, "done": 0,
            "current": None, "results": [], "error": None,
        }

    def start(self, groups, process_draft, inline: bool = False) -> None:
        with self._lock:
            if self._state["status"] == "running":
                raise EvolutionBusy("evolution is already running")
            total = sum(len(lessons) for _, lessons in groups)
            self._state = {
                "status": "running", "total": total, "done": 0,
                "current": None, "results": [], "error": None,
            }

        def run() -> None:
            failures = 0
            for draft_id, lessons in groups:
                with self._lock:
                    self._state["current"] = {
                        "draft_id": draft_id, "lesson_id": None,
                    }
                try:
                    result = self.runner.run_batch_now(
                        draft_id, lambda: process_draft(draft_id, lessons)
                    )
                except Exception as exc:  # noqa: BLE001
                    result = {
                        "draft_id": draft_id, "error": str(exc), "lessons": [],
                    }
                with self._lock:
                    self._state["results"].append(result)
                    self._state["done"] += len(lessons)
                    self._state["current"] = None
                if result.get("error") or any(
                    not item.get("ok") for item in result.get("lessons", [])
                ):
                    failures += 1
            with self._lock:
                self._state["status"] = "failed" if failures else "done"
                self._state["error"] = f"{failures} 个草稿失败" if failures else None

        if inline:
            run()
        else:
            threading.Thread(target=run, daemon=True).start()

    def status(self) -> dict:
        with self._lock:
            return copy.deepcopy(self._state)
```
