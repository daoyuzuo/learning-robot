```python
"""Draft state and files for the content studio."""

from __future__ import annotations

import json
import re
from pathlib import Path

from testbridge.learning.timeutil import iso, utcnow

SCHEMA = "studio/1"
ID_RE = re.compile(r"^[A-Za-z][A-Za-z0-9-]*$")
STEPS = ("outline", "research", "lessons", "validation", "promote")


class DraftError(Exception):
    def __init__(self, code: str, message: str) -> None:
        self.code = code
        self.message = message
        super().__init__(message)


class DraftNotFound(DraftError):
    def __init__(self, draft_id: str) -> None:
        super().__init__("draft_not_found", f"草稿不存在：{draft_id}")


def _step(kind: str) -> dict:
    return {"status": "pending", "error": None, "at": None}


class DraftStore:
    def __init__(self, root: Path) -> None:
        self.root = root

    def path(self, draft_id: str) -> Path:
        return self.root / draft_id

    def exists(self, draft_id: str) -> bool:
        return (self.path(draft_id) / "draft.json").is_file()

    def create(
        self,
        *,
        draft_id: str,
        topic: str,
        level: str,
        lesson_count: int,
        searches_per_lesson: int,
        model: str,
        prompt_version: str,
    ) -> dict:
        if not ID_RE.fullmatch(draft_id):
            raise DraftError("invalid_id", "草稿 ID 只能包含字母、数字和连字符")
        if self.exists(draft_id):
            raise DraftError("draft_exists", f"草稿已存在：{draft_id}")
        now = iso(utcnow())
        draft = {
            "schema": SCHEMA,
            "id": draft_id,
            "topic": topic,
            "level": level,
            "lesson_count": lesson_count,
            "searches_per_lesson": searches_per_lesson,
            "model": model,
            "prompt_version": prompt_version,
            "created_at": now,
            "updated_at": now,
            "usage": {"searches": 0, "llm_calls": 0},
            "steps": {
                "outline": _step("outline"),
                "research": {},
                "lessons": {},
                "validation": _step("validation"),
                "promote": _step("promote"),
            },
        }
        self.path(draft_id).mkdir(parents=True, exist_ok=True)
        self.save(draft)
        return draft

    def load(self, draft_id: str) -> dict:
        target = self.path(draft_id) / "draft.json"
        if not target.is_file():
            raise DraftNotFound(draft_id)
        return json.loads(target.read_text(encoding="utf-8"))

    def save(self, draft: dict) -> None:
        draft["updated_at"] = iso(utcnow())
        target = self.path(draft["id"]) / "draft.json"
        temporary = target.with_suffix(".json.tmp")
        temporary.write_text(
            json.dumps(draft, ensure_ascii=False, indent=2), encoding="utf-8"
        )
        temporary.replace(target)

    def list(self) -> list[dict]:
        if not self.root.is_dir():
            return []
        drafts = []
        for item in sorted(self.root.iterdir()):
            if (item / "draft.json").is_file():
                drafts.append(json.loads(
                    (item / "draft.json").read_text(encoding="utf-8")
                ))
        return sorted(drafts, key=lambda draft: draft["id"])

    def set_step(
        self,
        draft_id: str,
        step: str,
        *,
        status: str | None = None,
        error: str | None = None,
        extra: dict | None = None,
        lesson_id: str | None = None,
    ) -> dict:
        draft = self.load(draft_id)
        container = draft["steps"][step]
        if lesson_id is not None:
            entry = container.setdefault(lesson_id, _step(step))
        else:
            entry = container
        if status is not None:
            entry["status"] = status
            entry["at"] = iso(utcnow())
        entry["error"] = error
        if extra:
            entry.update(extra)
        self.save(draft)
        return draft

    def mark_interrupted(self) -> int:
        changed = 0
        for draft in self.list():
            dirty = False
            for step in STEPS:
                container = draft["steps"].get(step)
                entries = (
                    [container] if isinstance(container, dict) and "status" in container
                    else list((container or {}).values())
                )
                for entry in entries:
                    if entry.get("status") == "running":
                        entry["status"] = "failed"
                        entry["error"] = "interrupted"
                        changed += 1
                        dirty = True
            if dirty:
                self.save(draft)
        return changed

    def write(self, draft_id: str, relative: str, text: str) -> Path:
        target = self.path(draft_id) / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        temporary = target.with_suffix(target.suffix + ".tmp")
        temporary.write_text(text, encoding="utf-8")
        temporary.replace(target)
        return target

    def read(self, draft_id: str, relative: str) -> str:
        return (self.path(draft_id) / relative).read_text(encoding="utf-8")
```

```python
"""DeepSeek Search & Fetch client."""

from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Protocol

import httpx

from testbridge.tutor.config import TutorConfig

TRUST_BY_AUTHORITY = {4: "A", 3: "A", 2: "B", 1: "C", 0: "D", -1: "D"}


def trust_for(authority: int) -> str:
    return TRUST_BY_AUTHORITY.get(authority, "D")


class SearchError(Exception):
    """The search or fetch request failed."""


@dataclass(frozen=True)
class SearchHit:
    title: str
    url: str
    snippet: str
    summary: str
    content: str
    authority: int
    date: str | None = None


class SearchProvider(Protocol):
    def search(self, query: str, count: int = 10) -> list[SearchHit]: ...

    def fetch(self, url: str) -> str: ...


class DeepSeekSearch:
    def __init__(
        self,
        config: TutorConfig,
        transport: httpx.BaseTransport | None = None,
        retry_delay: float = 1.0,
    ) -> None:
        self.config = config
        self.retry_delay = retry_delay
        self._client = httpx.Client(timeout=config.timeout, transport=transport)

    def _post(self, path: str, payload: dict) -> dict:
        url = self.config.base_url.rstrip("/") + path
        last_error = "search failed"
        for attempt in range(2):
            try:
                response = self._client.post(
                    url,
                    headers={"Authorization": f"Bearer {self.config.api_key}"},
                    json=payload,
                )
                if response.status_code < 400:
                    return response.json()
                last_error = f"upstream status {response.status_code}"
                if response.status_code != 429 and response.status_code < 500:
                    break
            except (httpx.TimeoutException, httpx.HTTPError) as exc:
                last_error = type(exc).__name__
            if attempt == 0 and self.retry_delay:
                time.sleep(self.retry_delay)
        raise SearchError(last_error)

    def search(self, query: str, count: int = 10) -> list[SearchHit]:
        data = self._post(
            "/tools/search",
            {"query": query, "count": count, "need_content": True},
        )
        results = data.get("results") if isinstance(data, dict) else None
        if not isinstance(results, list):
            return []
        hits = []
        for item in results:
            if not isinstance(item, dict) or not item.get("url"):
                continue
            hits.append(SearchHit(
                title=str(item.get("title") or ""),
                url=str(item["url"]),
                snippet=str(item.get("snippet") or ""),
                summary=str(item.get("summary") or ""),
                content=str(item.get("content") or ""),
                authority=int(item.get("authority") or -1),
                date=item.get("date"),
            ))
        return hits

    def fetch(self, url: str) -> str:
        data = self._post("/tools/fetch", {"url": url})
        return str(data.get("content") or "") if isinstance(data, dict) else ""
```

"""Outline parsing, HTML rendering, and contract file writing."""

from __future__ import annotations

import html as html_lib
import json
import re
from dataclasses import dataclass
from html.parser import HTMLParser

import yaml

from testbridge.contract.sanitize import sanitize_fragment
from testbridge.studio.store import ID_RE, DraftStore

MAX_CONCEPTS = 8


class OutlineError(Exception):
    def __init__(self, message: str) -> None:
        self.message = message
        super().__init__(message)


@dataclass(frozen=True)
class LessonPlan:
    lesson_id: str
    title: str
    minutes: int
    objectives: list[str]
    queries: list[str]


def _check_id(value: str, label: str) -> None:
    if not isinstance(value, str) or not ID_RE.fullmatch(value):
        raise OutlineError(f"{label} 不是合法 ID：{value!r}")


def parse_outline_json(raw: str, lesson_count: int) -> dict:
    text = raw.strip()
    if text.startswith("```"):
        text = re.sub(r"^```(?:json)?\s*|\s*```$", "", text, flags=re.IGNORECASE)
    try:
        data = json.loads(text)
    except json.JSONDecodeError as exc:
        raise OutlineError(f"大纲不是合法 JSON：{exc.msg}") from None
    if not isinstance(data, dict):
        raise OutlineError("大纲必须是 JSON 对象")

    title = data.get("module_title")
    if not isinstance(title, str) or not title.strip():
        raise OutlineError("缺少 module_title")

    concepts = data.get("concepts")
    if not isinstance(concepts, list) or not 1 <= len(concepts) <= MAX_CONCEPTS:
        raise OutlineError("concepts 数量必须在 1–8 之间")
    concept_ids = []
    for concept in concepts:
        if not isinstance(concept, dict):
            raise OutlineError("concept 必须是对象")
        _check_id(concept.get("id"), "concept.id")
        concept_ids.append(concept["id"])
    if len(set(concept_ids)) != len(concept_ids):
        raise OutlineError("concept.id 必须唯一")
    for concept in concepts:
        for item in concept.get("prerequisites") or []:
            if item not in concept_ids:
                raise OutlineError(f"先修概念不存在：{item}")

    lessons = data.get("lessons")
    if not isinstance(lessons, list) or len(lessons) != lesson_count:
        raise OutlineError(f"课程数量必须是 {lesson_count}")
    lesson_ids = []
    for lesson in lessons:
        if not isinstance(lesson, dict):
            raise OutlineError("lesson 必须是对象")
        _check_id(lesson.get("id"), "lesson.id")
        lesson_ids.append(lesson["id"])
        if not lesson.get("objectives"):
            raise OutlineError(f"{lesson['id']} 缺少 objectives")
    if len(set(lesson_ids)) != len(lesson_ids):
        raise OutlineError("lesson.id 必须唯一")
    return data


def render_outline_html(module_title: str, lessons: list[LessonPlan]) -> str:
    sections = []
    for plan in lessons:
        objectives = "".join(
            f"<li>{html_lib.escape(item)}</li>" for item in plan.objectives
        )
        queries = "".join(
            f"<li>{html_lib.escape(item)}</li>" for item in plan.queries
        )
        sections.append(
            f'<section class="tb-outline-lesson" id="{html_lib.escape(plan.lesson_id)}">'
            f"<h2>{html_lib.escape(plan.title)}</h2>"
            f'<p class="tb-minutes">{plan.minutes}</p>'
            f'<ul class="tb-objectives">{objectives}</ul>'
            f'<ul class="tb-queries">{queries}</ul>'
            "</section>"
        )
    return (
        '<article class="tb-outline">'
        f'<h1 class="tb-module-title">{html_lib.escape(module_title)}</h1>'
        + "".join(sections)
        + "</article>"
    )


class _OutlineParser(HTMLParser):
    def __init__(self) -> None:
        super().__init__(convert_charrefs=True)
        self.module_title = ""
        self.lessons: list[dict] = []
        self._current: dict | None = None
        self._capture: str | None = None
        self._buffer: list[str] = []

    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        values = dict(attrs)
        classes = (values.get("class") or "").split()
        if tag == "section" and "tb-outline-lesson" in classes:
            self._current = {
                "id": values.get("id") or "", "title": "", "minutes": 20,
                "objectives": [], "queries": [],
            }
        elif tag == "h1" and "tb-module-title" in classes:
            self._capture = "title"
            self._buffer = []
        elif tag == "h2" and self._current is not None:
            self._capture = "lesson_title"
            self._buffer = []
        elif tag == "p" and "tb-minutes" in classes:
            self._capture = "minutes"
            self._buffer = []
        elif tag == "ul" and self._current is not None:
            self._capture = (
                "objectives" if "tb-objectives" in classes
                else "queries" if "tb-queries" in classes else None
            )
        elif tag == "li" and self._capture in {"objectives", "queries"}:
            self._buffer = []

    def handle_endtag(self, tag: str) -> None:
        if tag == "h1" and self._capture == "title":
            self.module_title = "".join(self._buffer).strip()
            self._capture = None
        elif tag == "h2" and self._capture == "lesson_title" and self._current is not None:
            self._current["title"] = "".join(self._buffer).strip()
            self._capture = None
        elif tag == "p" and self._capture == "minutes" and self._current is not None:
            try:
                self._current["minutes"] = int("".join(self._buffer).strip())
            except ValueError:
                self._current["minutes"] = 20
            self._capture = None
        elif tag == "li" and self._capture in {"objectives", "queries"}:
            value = "".join(self._buffer).strip()
            if value and self._current is not None:
                self._current[self._capture].append(value)
            self._buffer = []
        elif tag == "ul":
            self._capture = None
        elif tag == "section" and self._current is not None:
            self.lessons.append(self._current)
            self._current = None

    def handle_data(self, data: str) -> None:
        if self._capture:
            self._buffer.append(data)


def parse_outline_html(html: str) -> list[LessonPlan]:
    parser = _OutlineParser()
    parser.feed(html)
    parser.close()
    plans = []
    for item in parser.lessons:
        _check_id(item["id"], "lesson.id")
        plans.append(LessonPlan(
            lesson_id=item["id"],
            title=item["title"] or item["id"],
            minutes=item["minutes"] or 20,
            objectives=item["objectives"],
            queries=item["queries"],
        ))
    if not plans:
        raise OutlineError("大纲里没有找到课程")
    return plans


def write_outline(store: DraftStore, draft_id: str, parsed: dict) -> list[LessonPlan]:
    draft = store.load(draft_id)
    lessons = [
        LessonPlan(
            lesson_id=item["id"],
            title=str(item.get("title") or item["id"]),
            minutes=int(item.get("minutes") or 20),
            objectives=[str(value) for value in item.get("objectives") or []],
            queries=[str(value) for value in item.get("research_queries") or []],
        )
        for item in parsed["lessons"]
    ]
    concepts = parsed["concepts"]

    pack = {
        "schema": "tb/1",
        "id": draft_id,
        "version": "0.1.0",
        "title": parsed["module_title"],
        "description": draft["topic"],
        "language": "zh-CN",
        "license": "CC-BY-4.0",
        "authors": ["studio"],
        "created": draft["created_at"][:10],
    }
    module = {
        "schema": "tb/1",
        "id": "M01",
        "title": parsed["module_title"],
        "priority": "P1",
        "concepts": [item["id"] for item in concepts],
        "lessons": [plan.lesson_id for plan in lessons],
        "order": 1,
    }

    store.write(draft_id, "pack.yaml", yaml.safe_dump(
        pack, allow_unicode=True, sort_keys=False))
    store.write(draft_id, "concepts.yaml", yaml.safe_dump(
        {"schema": "tb/1", "concepts": concepts},
        allow_unicode=True, sort_keys=False))
    store.write(draft_id, "modules/M01/module.yaml", yaml.safe_dump(
        module, allow_unicode=True, sort_keys=False))
    store.write(
        draft_id, "outline.html",
        sanitize_fragment(render_outline_html(parsed["module_title"], lessons)),
    )

    for step in ("research", "lessons"):
        container = draft["steps"][step]
        draft["steps"][step] = {
            plan.lesson_id: container.get(plan.lesson_id, {
                "status": "pending", "error": None, "at": None,
            })
            for plan in lessons
        }
    store.save(draft)
    return lessons

"""Search orchestration and research dossier."""

from __future__ import annotations

import html as html_lib
import json
from collections.abc import Callable
from dataclasses import dataclass, field

from testbridge.studio.outline import LessonPlan
from testbridge.studio.search import SearchHit, SearchProvider, trust_for
from testbridge.studio.store import DraftStore

SITE_QUERY = "site:docs.ros.org OR site:docs.ros2.org"
CONTENT_LIMIT = 4000


@dataclass(frozen=True)
class ResearchResult:
    lesson_id: str
    queries: list[str]
    hits: list[SearchHit]
    errors: list[str] = field(default_factory=list)
    searches: int = 0


def build_queries(
    lesson: LessonPlan, concept_titles: list[str], budget: int
) -> list[str]:
    queries = list(lesson.queries)
    queries.append(lesson.title)
    queries.append(f"{lesson.title} {SITE_QUERY}")
    queries.extend(concept_titles)
    seen: list[str] = []
    for query in queries:
        if query and query not in seen:
            seen.append(query)
    return seen[:max(1, budget)]


def run_research(
    lesson: LessonPlan,
    concept_titles: list[str],
    search: SearchProvider,
    budget: int,
    progress: Callable[[int, int], None] | None = None,
) -> ResearchResult:
    queries = build_queries(lesson, concept_titles, budget)
    hits: list[SearchHit] = []
    seen: set[str] = set()
    errors: list[str] = []
    for index, query in enumerate(queries, start=1):
        try:
            for hit in search.search(query, count=8):
                if hit.url in seen:
                    continue
                seen.add(hit.url)
                hits.append(hit)
        except Exception as exc:  # noqa: BLE001 - keep research going
            errors.append(f"{query}: {type(exc).__name__}")
        if progress is not None:
            progress(index, len(queries))
    return ResearchResult(
        lesson_id=lesson.lesson_id, queries=queries, hits=hits,
        errors=errors, searches=len(queries),
    )


def render_research_html(lesson: LessonPlan, result: ResearchResult) -> str:
    rows = []
    for hit in result.hits:
        rows.append(
            f'<li class="tb-hit"><a href="{html_lib.escape(hit.url)}">'
            f"{html_lib.escape(hit.title or hit.url)}</a> "
            f'<span class="tb-trust">{trust_for(hit.authority)}</span>'
            f"<p>{html_lib.escape(hit.summary or hit.snippet)}</p></li>"
        )
    errors = "".join(
        f"<li>{html_lib.escape(item)}</li>" for item in result.errors
    )
    return (
        '<article class="tb-research">'
        f"<h1>调研档案：{html_lib.escape(lesson.title)}</h1>"
        f"<p>查询 {len(result.queries)} 次，命中 {len(result.hits)} 条。</p>"
        f'<ul class="tb-queries">'
        + "".join(f"<li>{html_lib.escape(q)}</li>" for q in result.queries)
        + "</ul>"
        f'<ul class="tb-hits">{"".join(rows)}</ul>'
        + (f'<ul class="tb-errors">{errors}</ul>' if errors else "")
        + "</article>"
    )


def write_research(
    store: DraftStore,
    draft_id: str,
    lesson: LessonPlan,
    result: ResearchResult,
) -> None:
    payload = {
        "lesson_id": result.lesson_id,
        "queries": result.queries,
        "searches": result.searches,
        "errors": result.errors,
        "hits": [
            {
                "title": hit.title, "url": hit.url, "snippet": hit.snippet,
                "summary": hit.summary, "content": hit.content[:CONTENT_LIMIT],
                "authority": hit.authority, "date": hit.date,
                "trust": trust_for(hit.authority),
            }
            for hit in result.hits
        ],
    }
    store.write(
        draft_id, f"research/{result.lesson_id}.json",
        json.dumps(payload, ensure_ascii=False, indent=2),
    )
    store.write(
        draft_id, f"research/{result.lesson_id}.html",
        render_research_html(lesson, result),
    )

"""Turn model output into contract files for one lesson."""

from __future__ import annotations

import json
import re
from dataclasses import dataclass, field

import yaml

from testbridge.contract.sanitize import sanitize_fragment
from testbridge.studio.outline import LessonPlan
from testbridge.studio.store import DraftStore

MODULE_ID = "M01"
ALLOWED_TRUST = {"A", "B"}
_FENCE_RE = re.compile(r"^```(?:json)?\s*|\s*```$", re.IGNORECASE)


class LessonError(Exception):
    def __init__(self, message: str) -> None:
        self.message = message
        super().__init__(message)


@dataclass(frozen=True)
class LessonOutput:
    manifest: dict
    html: str
    exercises: list[dict] = field(default_factory=list)
    resources: list[dict] = field(default_factory=list)


def parse_lesson_json(
    raw: str, lesson: LessonPlan, concept_ids: set[str]
) -> LessonOutput:
    text = _FENCE_RE.sub("", raw.strip()).strip()
    try:
        data = json.loads(text)
    except json.JSONDecodeError as exc:
        raise LessonError(f"写课输出不是合法 JSON：{exc.msg}") from None
    if not isinstance(data, dict):
        raise LessonError("写课输出必须是 JSON 对象")

    html = data.get("html")
    if not isinstance(html, str) or "<article" not in html:
        raise LessonError("缺少 HTML 正文")
    if re.search(r"<\s*script", html, re.IGNORECASE):
        raise LessonError("正文不得包含脚本")
    clean = sanitize_fragment(html)
    if "<article" not in clean:
        raise LessonError("正文净化后为空")

    manifest = data.get("manifest") if isinstance(data.get("manifest"), dict) else {}
    objectives = manifest.get("objectives") or lesson.objectives
    minutes = manifest.get("minutes") or lesson.minutes

    exercises = data.get("exercises") or []
    resources = data.get("resources") or []
    if not isinstance(exercises, list) or not isinstance(resources, list):
        raise LessonError("exercises / resources 必须是数组")

    exercise_ids = set()
    for exercise in exercises:
        if not isinstance(exercise, dict) or not exercise.get("id"):
            raise LessonError("练习缺少 id")
        if exercise["id"] in exercise_ids:
            raise LessonError(f"练习 id 重复：{exercise['id']}")
        exercise_ids.add(exercise["id"])
        for concept_id in exercise.get("concepts") or []:
            if concept_id not in concept_ids:
                raise LessonError(f"练习引用了不存在的概念：{concept_id}")

    for resource in resources:
        if not isinstance(resource, dict) or not resource.get("id"):
            raise LessonError("资源缺少 id")
        if resource.get("trust") not in ALLOWED_TRUST:
            raise LessonError(f"资源只有 A/B 级可用：{resource.get('id')}")
        for concept_id in resource.get("coverage") or []:
            if concept_id not in concept_ids:
                raise LessonError(f"资源覆盖了不存在的概念：{concept_id}")

    return LessonOutput(
        manifest={
            "title": manifest.get("title") or lesson.title,
            "minutes": int(minutes),
            "objectives": [str(item) for item in objectives],
        },
        html=clean,
        exercises=exercises,
        resources=resources,
    )


def _load_list(store: DraftStore, draft_id: str, relative: str, key: str) -> list:
    try:
        data = yaml.safe_load(store.read(draft_id, relative))
    except FileNotFoundError:
        return []
    if not isinstance(data, dict):
        return []
    return list(data.get(key) or [])


def _dump(data: dict) -> str:
    return yaml.safe_dump(data, allow_unicode=True, sort_keys=False)


def write_lesson(
    store: DraftStore,
    draft_id: str,
    lesson: LessonPlan,
    output: LessonOutput,
) -> None:
    existing_exercises = _load_list(store, draft_id, "exercises.yaml", "exercises")
    existing_resources = _load_list(store, draft_id, "resources.yaml", "resources")
    known_exercises = {item.get("id") for item in existing_exercises}
    known_resources = {item.get("id") for item in existing_resources}

    for exercise in output.exercises:
        if exercise["id"] in known_exercises:
            raise LessonError(f"练习 id 已存在：{exercise['id']}")
    for resource in output.resources:
        if resource["id"] in known_resources:
            raise LessonError(f"资源 id 已存在：{resource['id']}")

    module = yaml.safe_load(store.read(draft_id, f"modules/{MODULE_ID}/module.yaml"))
    module_concepts = list(module.get("concepts") or [])
    used_concepts: list[str] = []
    for exercise in output.exercises:
        for concept_id in exercise.get("concepts") or []:
            if concept_id not in used_concepts:
                used_concepts.append(concept_id)
    manifest_concepts = used_concepts or module_concepts
    target_level = str(store.load(draft_id).get("level") or "L2")

    manifest = {
        "schema": "tb/1",
        "id": lesson.lesson_id,
        "title": output.manifest["title"],
        "module": MODULE_ID,
        "concepts": manifest_concepts,
        "target_level": target_level,
        "minutes": output.manifest["minutes"],
        "objectives": output.manifest["objectives"],
        "prerequisites": [],
        "exercises": [item["id"] for item in output.exercises],
        "resources": [item["id"] for item in output.resources],
    }

    document = (
        '<script type="application/json" id="tb-manifest">\n'
        + json.dumps(manifest, ensure_ascii=False, indent=2)
        + "\n</script>\n"
        + output.html
    )

    store.write(
        draft_id, "exercises.yaml",
        _dump({"schema": "tb/1", "exercises": existing_exercises + output.exercises}),
    )
    store.write(
        draft_id, "resources.yaml",
        _dump({"schema": "tb/1", "resources": existing_resources + output.resources}),
    )
    store.write(draft_id, f"modules/{MODULE_ID}/lessons/{lesson.lesson_id}.html", document)

    lessons = list(module.get("lessons") or [])
    if lesson.lesson_id not in lessons:
        lessons.append(lesson.lesson_id)
        module["lessons"] = lessons
        store.write(
            draft_id, f"modules/{MODULE_ID}/module.yaml", _dump(module)
        )

"""Validation, resource verification, and promotion."""

from __future__ import annotations

import shutil
from dataclasses import dataclass, field
from pathlib import Path

import httpx
import yaml

from testbridge.contract.validator import validate_pack
from testbridge.studio.store import DraftStore

CONTENT_FILES = (
    "pack.yaml", "concepts.yaml", "exercises.yaml", "resources.yaml", "widgets.yaml",
)
CONTENT_DIRS = ("modules", "assets")


class PromoteError(Exception):
    def __init__(self, code: str, message: str) -> None:
        self.code = code
        self.message = message
        super().__init__(message)


@dataclass(frozen=True)
class ValidationSummary:
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not self.errors


def validate_draft(store: DraftStore, draft_id: str) -> ValidationSummary:
    report = validate_pack(store.path(draft_id))
    return ValidationSummary(
        errors=[f"{issue.code}: {issue.message}" for issue in report.errors],
        warnings=[f"{issue.code}: {issue.message}" for issue in report.warnings],
    )


def verify_resources(resources, client=None) -> list[str]:
    owned = client is None
    client = client or httpx.Client(timeout=10.0, follow_redirects=True)
    errors = []
    try:
        for resource in resources:
            resource_id = resource.get("id", "?")
            url = resource.get("url", "")
            try:
                response = client.get(url)
                if response.status_code >= 400:
                    errors.append(f"{resource_id}: HTTP {response.status_code}")
            except httpx.HTTPError as exc:
                errors.append(f"{resource_id}: {type(exc).__name__}")
    finally:
        if owned:
            client.close()
    return errors


def promote(
    store: DraftStore,
    draft_id: str,
    packs_root: Path,
    *,
    force: bool = False,
    verifier=None,
) -> Path:
    source = store.path(draft_id)
    target = packs_root / draft_id
    if target.exists() and not force:
        raise PromoteError("pack_exists", f"内容包已存在：{draft_id}")

    summary = validate_draft(store, draft_id)
    if not summary.ok:
        raise PromoteError("invalid_pack", "；".join(summary.errors[:5]))

    resources_path = source / "resources.yaml"
    if resources_path.is_file():
        data = yaml.safe_load(resources_path.read_text(encoding="utf-8")) or {}
        check = verifier or verify_resources
        errors = check(data.get("resources") or [])
        if errors:
            raise PromoteError("resource_unreachable", "；".join(errors[:5]))

    target.mkdir(parents=True, exist_ok=True)
    for name in CONTENT_FILES:
        item = source / name
        if item.is_file():
            shutil.copy2(item, target / name)
    for name in CONTENT_DIRS:
        item = source / name
        if item.is_dir():
            shutil.copytree(item, target / name, dirs_exist_ok=True)
    return target
```

"""Run studio steps in background threads with persisted status."""

from __future__ import annotations

import threading
from collections.abc import Callable

from testbridge.studio.store import DraftStore

KINDS = ("outline", "research", "lessons", "validate", "promote")
STEP_FOR_KIND = {
    "outline": "outline", "research": "research", "lessons": "lessons",
    "validate": "validation", "promote": "promote",
}


class JobBusy(Exception):
    """A job for the same draft step is already running."""


class JobRunner:
    def __init__(self, store: DraftStore, inline: bool = False) -> None:
        self.store = store
        self.inline = inline
        self._running: set[tuple[str, str, str | None]] = set()
        self._lock = threading.Lock()

    def start(
        self,
        draft_id: str,
        kind: str,
        fn: Callable[[], None],
        *,
        lesson_id: str | None = None,
    ) -> None:
        if kind not in KINDS:
            raise ValueError(f"unknown job kind: {kind}")
        key = (draft_id, kind, lesson_id)
        with self._lock:
            if key in self._running:
                raise JobBusy(f"{kind} is already running")
            self._running.add(key)

        self.store.set_step(
            draft_id, STEP_FOR_KIND[kind], status="running", lesson_id=lesson_id
        )

        def run() -> None:
            try:
                fn()
            except Exception as exc:  # noqa: BLE001 - status carries the error
                self.store.set_step(
                    draft_id, STEP_FOR_KIND[kind], status="failed",
                    error=str(exc), lesson_id=lesson_id,
                )
            else:
                self.store.set_step(
                    draft_id, STEP_FOR_KIND[kind], status="done", lesson_id=lesson_id
                )
            finally:
                with self._lock:
                    self._running.discard(key)

        if self.inline:
            run()
        else:
            threading.Thread(target=run, daemon=True).start()
```

from __future__ import annotations

from pathlib import Path

from fastapi import APIRouter, Form, HTTPException, Request
from fastapi.responses import HTMLResponse, RedirectResponse
from pydantic import BaseModel

from testbridge.studio.jobs import JobBusy, JobRunner
from testbridge.studio.outline import OutlineError, parse_outline_html, write_outline
from testbridge.studio.prompts import (
    LESSON_SYSTEM, OUTLINE_SYSTEM, render_lesson_prompt, render_outline_prompt,
)
from testbridge.studio.search import SearchProvider, trust_for
from testbridge.studio.store import DraftError, DraftStore
from testbridge.tutor.client import LlmClient
from testbridge.tutor.config import TutorConfig, missing_vars


class OutlineBody(BaseModel):
    html: str


def mount_studio(
    app,
    templates,
    *,
    drafts_root: Path,
    packs_root: Path,
    config: TutorConfig | None,
    llm: LlmClient | None,
    search: SearchProvider | None,
    inline: bool = False,
) -> None:
    store = DraftStore(drafts_root)
    store.mark_interrupted()
    runner = JobRunner(store, inline=inline)
    app.state.studio_store = store
    router = APIRouter()

    def ctx(request: Request, **kwargs):
        return {
            "request": request,
            "missing": missing_vars() if config is None else [],
            "enabled": config is not None and llm is not None and search is not None,
            **kwargs,
        }

    def require_enabled() -> None:
        if config is None or llm is None or search is None:
            raise HTTPException(
                status_code=503,
                detail={"code": "tutor_disabled", "message": "未配置模型服务"},
            )

    def draft_or_404(draft_id: str) -> dict:
        if not store.exists(draft_id):
            raise HTTPException(
                status_code=404,
                detail={"code": "draft_not_found", "message": "草稿不存在"},
            )
        return store.load(draft_id)

    def concepts_text(draft: dict) -> str:
        import yaml

        data = yaml.safe_load(store.read(draft["id"], "concepts.yaml"))
        return "\n".join(
            f"- {item['id']} {item.get('title', '')}" for item in data["concepts"]
        )

    def generate_outline(draft_id: str) -> None:
        draft = store.load(draft_id)
        raw = llm.complete(
            OUTLINE_SYSTEM,
            render_outline_prompt(draft["topic"], draft["level"], draft["lesson_count"]),
        )
        parsed = outline_writer.parse_outline_json(raw, draft["lesson_count"])
        write_outline(store, draft_id, parsed)

    def generate_research(draft_id: str, lesson_id: str) -> None:
        draft = store.load(draft_id)
        plans = parse_outline_html(store.read(draft_id, "outline.html"))
        plan = next((item for item in plans if item.lesson_id == lesson_id), None)
        if plan is None:
            raise OutlineError(f"课程不存在：{lesson_id}")
        import yaml

        data = yaml.safe_load(store.read(draft_id, "concepts.yaml"))
        titles = [item.get("title", "") for item in data["concepts"]]
        result = research_runner.run_research(
            plan, titles, search, draft["searches_per_lesson"]
        )
        research_runner.write_research(store, draft_id, plan, result)
        draft["usage"]["searches"] += result.searches
        store.save(draft)

    def generate_lesson(draft_id: str, lesson_id: str) -> None:
        import yaml

        draft = store.load(draft_id)
        plans = parse_outline_html(store.read(draft_id, "outline.html"))
        plan = next((item for item in plans if item.lesson_id == lesson_id), None)
        if plan is None:
            raise OutlineError(f"课程不存在：{lesson_id}")
        research_path = store.path(draft_id) / f"research/{lesson_id}.json"
        research_text = (
            research_path.read_text(encoding="utf-8") if research_path.is_file() else ""
        )
        data = yaml.safe_load(store.read(draft_id, "concepts.yaml"))
        concept_ids = {item["id"] for item in data["concepts"]}
        raw = llm.complete(
            LESSON_SYSTEM,
            render_lesson_prompt(
                store.read(draft_id, "outline.html"), concepts_text(draft), research_text
            ),
        )
        output = lesson_writer.parse_lesson_json(raw, plan, concept_ids)
        lesson_writer.write_lesson(store, draft_id, plan, output)
        draft["usage"]["llm_calls"] += 1
        store.save(draft)

    @router.get("/studio", response_class=HTMLResponse)
    def studio_list(request: Request):
        return templates.TemplateResponse(
            request=request, name="studio_list.html",
            context=ctx(request, drafts=store.list()),
        )

    @router.post("/studio")
    def studio_create(
        request: Request,
        draft_id: str = Form(...),
        topic: str = Form(...),
        level: str = Form("L2"),
        lesson_count: int = Form(3),
        searches_per_lesson: int = Form(20),
    ):

            store.create(
                draft_id=draft_id, topic=topic, level=level,
                lesson_count=lesson_count, searches_per_lesson=searches_per_lesson,
                model=config.model if config else "unconfigured",
                prompt_version="studio/1",
            )
        except DraftError as exc:
            raise HTTPException(
                status_code=400, detail={"code": exc.code, "message": exc.message}
            ) from None
        if config is not None and llm is not None:
            runner.start(draft_id, "outline", lambda: generate_outline(draft_id))
        return RedirectResponse(url=f"/studio/{draft_id}", status_code=303)

    @router.get("/studio/{draft_id}", response_class=HTMLResponse)
    def studio_draft(request: Request, draft_id: str):
        draft = draft_or_404(draft_id)
        outline_path = store.path(draft_id) / "outline.html"
        return templates.TemplateResponse(
            request=request, name="studio_draft.html",
            context=ctx(
                request, draft=draft,
                outline_html=(outline_path.read_text(encoding="utf-8")
                              if outline_path.is_file() else ""),
                lessons=[
                    plan.lesson_id
                    for plan in (parse_outline_html(
                        outline_path.read_text(encoding="utf-8")
                    ) if outline_path.is_file() else [])
                ],
            ),
        )

    @router.get("/api/studio/{draft_id}")
    def studio_status(draft_id: str):
        draft = draft_or_404(draft_id)
        return {
            "id": draft["id"], "steps": draft["steps"], "usage": draft["usage"],
        }

    @router.post("/api/studio/{draft_id}/outline")
    def regenerate_outline(draft_id: str):
        draft_or_404(draft_id)
        require_enabled()
        try:
            runner.start(draft_id, "outline", lambda: generate_outline(draft_id))
        except JobBusy:
            raise HTTPException(
                status_code=409, detail={"code": "step_running", "message": "任务进行中"}
            ) from None
        return {"started": True}

    @router.put("/api/studio/{draft_id}/outline")
    def save_outline(draft_id: str, body: OutlineBody):
        draft_or_404(draft_id)
        clean = sanitize_fragment(body.html)
        try:
            plans = parse_outline_html(clean)
        except OutlineError as exc:
            raise HTTPException(
                status_code=400,
                detail={"code": "invalid_outline", "message": exc.message},
            ) from None
        store.write(draft_id, "outline.html", clean)
        draft = store.load(draft_id)
        for step in ("research", "lessons"):
            container = draft["steps"][step]
            draft["steps"][step] = {
                plan.lesson_id: container.get(plan.lesson_id, {
                    "status": "pending", "error": None, "at": None,
                })
                for plan in plans
            }
        store.save(draft)
        return {"lessons": [plan.lesson_id for plan in plans]}

    @router.post("/api/studio/{draft_id}/research")
    def start_research(draft_id: str, body: dict | None = None):
        draft_or_404(draft_id)
        require_enabled()
        lesson_id = (body or {}).get("lesson_id")
        targets = [lesson_id] if lesson_id else list(store.load(draft_id)["steps"]["research"])
        try:
            for target in targets:
                runner.start(
                    draft_id, "research",
                    lambda target=target: generate_research(draft_id, target),
                    lesson_id=target,
                )
        except JobBusy:
            raise HTTPException(
                status_code=409, detail={"code": "step_running", "message": "任务进行中"}
            ) from None
        return {"started": targets}

    @router.post("/api/studio/{draft_id}/lessons")
    def start_lessons(draft_id: str, body: dict | None = None):
        draft_or_404(draft_id)
        require_enabled()
        lesson_id = (body or {}).get("lesson_id")
        targets = [lesson_id] if lesson_id else list(store.load(draft_id)["steps"]["lessons"])
        try:
            for target in targets:
                runner.start(
                    draft_id, "lessons",
                    lambda target=target: generate_lesson(draft_id, target),
                    lesson_id=target,
                )
        except JobBusy:
            raise HTTPException(
                status_code=409, detail={"code": "step_running", "message": "任务进行中"}
            ) from None
        return {"started": targets}

    @router.post("/api/studio/{draft_id}/validate")
    def run_validation(draft_id: str):
        draft_or_404(draft_id)
        summary = validate_draft(store, draft_id)
        store.set_step(
            draft_id, "validation", status="done" if summary.ok else "failed",
            error=None if summary.ok else summary.errors[0],
            extra={"issues": summary.errors + summary.warnings},
        )
        return {
            "ok": summary.ok, "errors": summary.errors, "warnings": summary.warnings,
        }

    @router.post("/api/studio/{draft_id}/promote")
    def run_promote(draft_id: str, body: dict | None = None):
        draft_or_404(draft_id)
        force = bool((body or {}).get("force"))
        try:
            target = promote(store, draft_id, packs_root, force=force)
        except PromoteError as exc:
            store.set_step(draft_id, "promote", status="failed", error=exc.message)
            raise HTTPException(
                status_code=400, detail={"code": exc.code, "message": exc.message}
            ) from None
        store.set_step(draft_id, "promote", status="done", extra={"path": str(target)})
        return {"path": str(target)}

    app.include_router(router)

(function () {
  const root = document.getElementById("studio");
  if (!root) return;
  const draftId = root.dataset.draft;
  const enabled = root.dataset.enabled === "true";
  const error = root.querySelector(".tb-error");
  let timer = null;

  function showError(message) {
    error.hidden = false;
    error.textContent = message;
  }

  async function call(url, options) {
    const response = await fetch(url, options);
    const payload = await response.json().catch(() => ({}));
    if (!response.ok) {
      throw new Error(payload.detail ? (payload.detail.message || payload.detail) : "请求失败");
    }
    return payload;
  }

  function paint(status) {
    document.querySelectorAll("[data-status-for]").forEach((node) => {
      node.textContent = (status.steps[node.dataset.statusFor] || {}).status || "pending";
    });
  }

  async function poll() {
    try {
      const payload = await call("/api/studio/" + draftId);
      paint(payload);
      const busy = JSON.stringify(payload.steps).includes('"running"');
      if (!busy && timer) {
        clearInterval(timer);
        timer = null;
      }
    } catch (err) {
      showError(err.message);
    }
  }

  function startPolling() {
    if (!timer) timer = setInterval(poll, 2000);
  }

  root.querySelectorAll("[data-action]").forEach((button) => {
    button.addEventListener("click", async () => {
      if (!enabled) return;
      error.hidden = true;
      const action = button.dataset.action;
      const lesson = button.dataset.lesson || null;
      try {
        if (action === "save-outline") {
          const editor = root.querySelector(".tb-outline-editor");
          await call("/api/studio/" + draftId + "/outline", {
            method: "PUT",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ html: editor.innerHTML }),
          });
          location.reload();
          return;
        }
        if (action === "regen-outline") {
          await call("/api/studio/" + draftId + "/outline", { method: "POST" });
        } else if (action === "research") {
          await call("/api/studio/" + draftId + "/research", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ lesson_id: lesson }),
          });
        } else if (action === "lesson") {
          await call("/api/studio/" + draftId + "/lessons", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ lesson_id: lesson }),
          });
        } else if (action === "validate") {
          const payload = await call("/api/studio/" + draftId + "/validate", { method: "POST" });
          root.querySelector(".tb-issues").textContent =
            [...payload.errors, ...payload.warnings].join("\n") || "校验通过";
        } else if (action === "promote") {
          const force = root.querySelector(".tb-force").checked;
          const payload = await call("/api/studio/" + draftId + "/promote", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ force }),
          });
          root.querySelector(".tb-issues").textContent = "已写入 " + payload.path;
        }
        startPolling();
        poll();
      } catch (err) {
        showError(err.message);
      }
    });
  });
})();
